cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GameEngineRenderer VERSION 0.1.0 LANGUAGES C CXX)
    enable_testing()
endif()
if(NOT TARGET engine::core)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../Core" "${CMAKE_CURRENT_BINARY_DIR}/core")
endif()
option(ENGINE_RENDERER_BUILD_TESTS "Build render snapshot acceptance tests" ON)
add_library(engine_renderer STATIC src/RenderSnapshot.cpp)
add_library(engine::renderer ALIAS engine_renderer)
target_compile_features(engine_renderer PUBLIC cxx_std_20)
set_target_properties(engine_renderer PROPERTIES CXX_EXTENSIONS NO)
target_include_directories(engine_renderer PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_include_directories(engine_renderer SYSTEM PRIVATE "${ENGINE_DEPENDENCY_ROOT}/glm")
target_link_libraries(engine_renderer PUBLIC engine::core)
if(MSVC)
    target_compile_options(engine_renderer PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(engine_renderer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ENGINE_RENDERER_BUILD_TESTS)
    enable_testing()
    add_executable(engine_renderer_tests tests/RenderSnapshotTests.cpp)
    target_link_libraries(engine_renderer_tests PRIVATE engine::renderer)
    if(MSVC)
        target_compile_options(engine_renderer_tests PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(engine_renderer_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME engine.renderer.snapshot COMMAND engine_renderer_tests)
    set_tests_properties(engine.renderer.snapshot PROPERTIES TIMEOUT 60)
endif()
