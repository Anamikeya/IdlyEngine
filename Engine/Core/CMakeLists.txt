cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GameEngineCore VERSION 0.1.0 LANGUAGES C CXX)
    enable_testing()
endif()
set(ENGINE_DEPENDENCY_ROOT "" CACHE PATH "Read-only shared dependency source directory")
if(NOT ENGINE_DEPENDENCY_ROOT)
    message(FATAL_ERROR "Set ENGINE_DEPENDENCY_ROOT to the existing resources directory")
endif()
option(ENGINE_CORE_BUILD_TESTS "Build core acceptance tests" ON)
option(ENGINE_CORE_BUILD_EXAMPLE "Build headless example" ON)
get_filename_component(ENGINE_MODULE_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
foreach(required flecs/distr/flecs.c flecs/distr/flecs.h glm/glm/glm.hpp spdlog/include/spdlog/spdlog.h)
    if(NOT EXISTS "${ENGINE_DEPENDENCY_ROOT}/${required}")
        message(FATAL_ERROR "Missing audited source: ${ENGINE_DEPENDENCY_ROOT}/${required}")
    endif()
endforeach()
if(NOT TARGET engine_flecs)
    add_library(engine_flecs STATIC "${ENGINE_DEPENDENCY_ROOT}/flecs/distr/flecs.c")
    target_include_directories(engine_flecs SYSTEM PUBLIC "${ENGINE_DEPENDENCY_ROOT}/flecs/distr")
    target_compile_definitions(engine_flecs PUBLIC flecs_STATIC)
    set_target_properties(engine_flecs PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
    if(WIN32)
        target_link_libraries(engine_flecs PUBLIC ws2_32 wsock32 dbghelp)
    else()
        find_package(Threads REQUIRED)
        target_link_libraries(engine_flecs PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
    endif()
endif()
add_library(engine_core STATIC
    src/Application.cpp src/Diagnostics.cpp
    "${ENGINE_MODULE_ROOT}/ECS/src/World.cpp"
    "${ENGINE_MODULE_ROOT}/Scene/src/Scene.cpp"
    "${ENGINE_MODULE_ROOT}/Scene/src/Serialization.cpp"
    "${ENGINE_MODULE_ROOT}/Commands/src/CommandBus.cpp"
)
add_library(engine::core ALIAS engine_core)
target_compile_features(engine_core PUBLIC cxx_std_20)
set_target_properties(engine_core PROPERTIES CXX_EXTENSIONS NO)
foreach(module Core ECS Scene Commands)
    target_include_directories(engine_core PUBLIC "${ENGINE_MODULE_ROOT}/${module}/include")
endforeach()
target_include_directories(engine_core SYSTEM PRIVATE
    "${ENGINE_DEPENDENCY_ROOT}/glm" "${ENGINE_DEPENDENCY_ROOT}/spdlog/include")
target_link_libraries(engine_core PUBLIC engine_flecs)
if(MSVC)
    target_compile_options(engine_core PRIVATE /W4 /WX /permissive- /utf-8 /Zc:__cplusplus)
else()
    target_compile_options(engine_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ENGINE_CORE_BUILD_EXAMPLE)
    add_executable(engine_headless examples/Headless.cpp)
    target_link_libraries(engine_headless PRIVATE engine::core)
endif()
if(ENGINE_CORE_BUILD_TESTS)
    enable_testing()
    add_executable(engine_core_tests tests/CoreTests.cpp)
    target_link_libraries(engine_core_tests PRIVATE engine::core)
    if(MSVC)
        target_compile_options(engine_core_tests PRIVATE /W4 /WX /permissive- /utf-8)
    endif()
    add_test(NAME engine.core.acceptance COMMAND engine_core_tests)
    set_tests_properties(engine.core.acceptance PROPERTIES TIMEOUT 60)
endif()
